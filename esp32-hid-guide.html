HID 键盘输入器

01

硬件要求

模组
乐鑫 ESP32-S3-WROOM-1-N16R8，任意兼容官方 ESP32-S3-DevKitC-1 的开发板均可
规格
建议选已焊接排针版，方便反复插拔和按 BOOT / EN
为什么
N16 = 16MB Flash 存长文本；R8 = 8MB OPI PSRAM。别用 N8R2 / N4R2，分区和 PSRAM 参数都要跟着改
  • 板子有两个 Type-C：右口 UART（下载 + 调试），左口 USB（HID 输出）
  • 附带线必须是数据线，纯充电线会导致后续查不到端口
  • macOS 免驱，插上就能用（CH343 芯片需另装 WCH 驱动）

const char* STA_SSID = "YOUR_WIFI_SSID";
const char* STA_PASS = "YOUR_WIFI_PASSWORD";
const char* AP_SSID = "ESP32_HID_KEYBOARD";   // 同房间多台必须改成互不相同
const char* AP_PASS = "12345678";             // 建议一并改掉
const char* MDNS_NAME = "esp32-hid";          // 多台时加后缀，如 esp32-hid-neo
多人同房间必须改名。默认 AP 名和 192.168.4.1 所有人都一样，连错别人的板子毫无感知 —— 你在别人的板子上配了一通文本，自己这块没反应。各自改成 ESP32_HID_你的名字，AP 密码、mDNS 名也一起改，板子上贴个标签。
两个 YOUR_ 占位符不替换也能跑，但板子连不上热点，10 秒后退到 AP 模式（ESP32_HID_KEYBOARD / 12345678 / 192.168.4.1）。建议用手机热点，公司 WiFi 常开 802.1X 或客户端隔离，连不上也 ping 不通。

String htmlEscape(const String& s) {
  String out;
  out.reserve(s.length() * 1.2);
  for (size_t i = 0; i < s.length(); i++) {
    char c = s[i];
    switch (c) {
      case '&':  out += "&amp;";  break;
      case '"':  out += "&quot;"; break;
      case '<':  out += "&lt;";   break;
      case '>':  out += "&gt;";   break;
      default:   out += c;        break;
    }
  }
  return out;
}

void releaseAllKeys() {
  Keyboard.releaseAll();
  keyHeld = false;
}

bool isSupportedChar(char c) {
  if ((c >= 'A' && c <= 'Z') || (c >= 'a' && c <= 'z') || (c >= '0' && c <= '9')) return true;
  switch(c) {
    case '!': case '@': case '#': case '$': case '%': case '^': case '&': case '*':
    case '(': case ')': case '_': case '+': case '-': case '=': case '[': case ']':
    case '{': case '}': case ';': case '\'': case ':': case '"': case ',': case '.':
    case '/': case '<': case '>': case '?': case '\\': case '|': case '`': case '~':
    case ' ': case '\n': case '\r': return true;
    default: return false;
  }
}

bool scanFirstUnsupported(const String &txt) {
  errLine = 1;
  errCol = 1;
  for (size_t i = 0; i < txt.length(); i++) {
    char c = txt[i];
    if (c == '\n') {
      errLine++;
      errCol = 1;
      continue;
    }
    if (!isSupportedChar(c)) {
      return true;
    }
    errCol++;
  }
  errLine = 0;
  errCol = 0;
  return false;
}

void startTyping() {
  if (inputText.length() == 0) return;
  releaseAllKeys();
  typeIndex = 0;
  isTyping = true;
  isPaused = false;
  keyHeld = false;
  typeTimer = millis() - CHAR_DELAY;  // 回拨一个间隔，让首字符立即触发
}

void stopTyping() {
  isTyping = false;
  isPaused = false;
  releaseAllKeys();
}

void processTyping() {
  if (!isTyping || isPaused) return;

  unsigned long now = millis();

  if (typeIndex >= (int)inputText.length()) {
    isTyping = false;
    releaseAllKeys();
    return;
  }

  if (!keyHeld) {
    // 字符间隔门禁：上一个字符释放后必须等够 CHAR_DELAY 才按下一个
    if (now - typeTimer < (unsigned long)CHAR_DELAY) return;

    char c = inputText[typeIndex];

    // \r 已在存/取时归一化，这里再兜一层：残留 \r 直接跳过，避免重复回车
    if (c == '\r') {
      typeIndex++;
      typeTimer = now;
      return;
    }

    if (!isSupportedChar(c)) {
      typeIndex++;
      typeTimer = now;
      return;
    }

    if (c >= 'A' && c <= 'Z') {
      Keyboard.press(KEY_LEFT_SHIFT);
      Keyboard.press(c);
    }
    else if(c == '~') {
      Keyboard.press(KEY_LEFT_SHIFT);
      Keyboard.press('`');
    }
    else if(c == '|') {
      Keyboard.press(KEY_LEFT_SHIFT);
      Keyboard.press('\\');
    }
    else if(c == '\n') {
      Keyboard.press(KEY_RETURN);
    }
    else {
      Keyboard.press(c);
    }
    keyHeld = true;
    typeTimer = now;
  } else {
    if (now - typeTimer >= (unsigned long)HOLD_DELAY) {
      releaseAllKeys();
      keyHeld = false;
      typeIndex++;
      typeTimer = now;
    }
  }
}

// 统一换行为 \n。textarea 提交的是 \r\n，若 \r 和 \n 都发 RETURN 会变成每行多一个空行
void normalizeNewlines(String &s) {
  if (s.indexOf('\r') < 0) return;
  s.replace("\r\n", "\n");
  s.replace("\r", "\n");
}

void loadText() {
  if (!LittleFS.begin(true)) {
    Serial.println("LittleFS mount failed");
    inputText = "Hello NEO";
    errLine = 0; errCol = 0;
    return;
  }

  File file = LittleFS.open("/text.txt", "r");
  if (file) {
    inputText = file.readString();
    file.close();
    normalizeNewlines(inputText);
    Serial.printf("Loaded text, len: %d\n", inputText.length());
  } else {
    inputText = "Hello NEO";
    Serial.println("No saved text, use default");
  }
  scanFirstUnsupported(inputText);
}

bool saveText(String t) {
  if (!LittleFS.begin()) {
    Serial.println("LittleFS mount failed");
    return false;
  }

  normalizeNewlines(t);

  File file = LittleFS.open("/text.txt", "w");
  if (!file) {
    Serial.println("Cannot open file for write");
    return false;
  }

  size_t written = file.print(t);
  file.close();

  if (written == t.length()) {
    inputText = t;
    scanFirstUnsupported(inputText);
    Serial.printf("Saved, written %d bytes\n", written);
    return true;
  } else {
    Serial.printf("Save incomplete, want %d, wrote %d\n", t.length(), written);
    return false;
  }
}

void handleRoot() {
  String p = PAGE;
  p.replace("%TEXT%", htmlEscape(inputText));
  p.replace("%ERR_LINE%", String(errLine));
  p.replace("%ERR_COL%", String(errCol));

  String runStatus = "Idle";
  if(isTyping && isPaused) runStatus = "Paused";
  else if(isTyping) runStatus = "Typing";
  p.replace("%RUN_STATUS%", runStatus);

  String tipHtml = "";
  if (server.hasArg("status")) {
    if (server.arg("status") == "ok") {
      tipHtml = "<div class='status-ok'>✅ Text saved</div>";
    } else if (server.arg("status") == "fail") {
      tipHtml = "<div class='status-fail'>❌ Save failed</div>";
    }
  }
  p.replace("%TIP%", tipHtml);

  String warnHtml = "";
  if(errLine > 0){
    warnHtml = "<div class='warn'>⚠ HID unsupported character found at line " + String(errLine) + ", column " + String(errCol) + ", it will be skipped when typing "
    + "<button class='btn-locate' onclick='goLine()'>Locate</button></div>";
  }
  p.replace("%WARN%", warnHtml);

  server.send(200, "text/html; charset=utf-8", p);
}

void handleSave() {
  bool success = false;
  if (server.hasArg("text")) {
    success = saveText(server.arg("text"));
  }
  String redirectUrl = success ? "/?status=ok" : "/?status=fail";
  server.sendHeader("Location", redirectUrl);
  server.send(302, "text/plain", "");
}

void handleSend() {
  startTyping();
  server.sendHeader("Location", "/");
  server.send(302, "text/plain", "");
}

void handleStop() {
  stopTyping();
  server.sendHeader("Location", "/");
  server.send(302, "text/plain", "");
}

void initWiFi() {
  Serial.begin(115200);
  WiFi.mode(WIFI_STA);
  WiFi.begin(STA_SSID, STA_PASS);

  int t = 0;
  while (WiFi.status() != WL_CONNECTED && t < 10000) {
    delay(500);
    t += 500;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("STA OK, IP: " + WiFi.localIP().toString());
  } else {
    WiFi.mode(WIFI_AP);
    WiFi.softAP(AP_SSID, AP_PASS);
    Serial.println("AP mode, IP: 192.168.4.1");
  }
  MDNS.begin(MDNS_NAME);
  Serial.println("mDNS: " + String(MDNS_NAME) + ".local");
}

void setup() {
  pinMode(BTN_BOOT, INPUT_PULLUP);
  USB.begin();
  Keyboard.begin();
  loadText();
  initWiFi();

  server.on("/", handleRoot);
  server.on("/save", HTTP_POST, handleSave);
  server.on("/send", HTTP_POST, handleSend);
  server.on("/stop", HTTP_POST, handleStop);

  server.begin();
}

void loop() {
  server.handleClient();
  processTyping();

  unsigned long now = millis();
  bool btnPressed = (digitalRead(BTN_BOOT) == LOW);

  // 单击触发，防抖
  if (btnPressed && (now - lastPress > debounce)) {
    lastPress = now;
    if(isTyping) {
      isPaused = !isPaused;
      Serial.printf("Paused = %d\n", isPaused);
    } else {
      startTyping();
    }
  }
}

编译 + 烧录

Terminal · 参数一项都不要省 
cd ~/esp32_hid_key

arduino-cli compile --output-dir=build \
  --fqbn esp32:esp32:esp32s3 \
  --build-property build.partitions=default_16MB \
  --build-property build.flash_size=16MB \
  --build-property build.psram_type=opi \
  --build-property build.usb_cdc_on_boot=0 \
  --build-property build.flashmode=qio \
  --build-property build.flashfreq=80m \
  --build-property "build.extra_flags=-DHTTP_MAX_POST_SIZE=65536" .

arduino-cli board list
arduino-cli upload -p /dev/cu.wchusbserial1420 --fqbn esp32:esp32:esp32s3 --input-dir=build .
  • psram_type=opi 写错（写成 qspi）→ PSRAM 起不来、boot loop
  • usb_cdc_on_boot=0 → 原生 USB 不再枚举成串口，只做 HID。这是目标机认不到键盘的头号原因
  • 端口名以 board list 输出为准，可能是 cu.wchusbserial* / cu.usbserial-* / cu.usbmodem*

进下载模式

  1. 数据线插板子右侧 UART 口（左口是 HID 口，插它查不到串口）
  2. 按住 BOOT 不放 → 点一下 EN → 松开 EN → 保持 BOOT 约 1 秒再松开
  3. 跑 upload，成功后按一下 EN 复位，拔掉右口
报 Failed to connect 就是没进下载模式，回第 2 步重来，别怀疑代码。
06

怎么用

  1. 右口插电 → 手机/Mac 连同一个热点 → 浏览器开 esp32-hid.local（打不开就用串口日志里的 IP）
  2. 粘贴文本 → Save Text，写进 Flash，断电不丢
  3. 拔右口 → 左口插目标机 → 点目标输入框 → 按一下 BOOT 开始敲
操作效果
BOOT 单击空闲时开始输入；输入中暂停 / 继续
Stop Typing网页上点，立即中止并放开所有按键
Send Now从头重打一遍，不用重新配文本
远程 VPC / 云桌面：客户端支持 USB 重定向就把 HID 设备重定向进去；不支持就让 VPC 窗口全屏并保持焦点 —— 中途切窗口，字符会打到本地。
07

动手前必须知道的

只打 ASCII。中文、日文、emoji、Tab、方向键、Ctrl 组合键都不支持。固件是逐字符映射 US 键盘 scancode，没有输入法能力。
目标机必须是英文输入法。停在微软拼音上，即使敲的是 ASCII 也会进候选框变成中文。开工前 Shift 或 Win+Space 切过去。
  • ESP32 只支持 2.4GHz WiFi。手机热点开了「5GHz 优先」可能连不上
  • 漏字符：远程桌面 / VPC 里把 CHAR_DELAY 从 80 调到 120~200（约 12~20 字/秒）
  • macOS 首次插入会弹键盘类型识别向导并抢焦点，按 Esc 关掉，之后不再弹
  • 文本明文存在板子 Flash 里，用完把网页上的文本清空再 Save
  • 确认下 HID 外设接入符合所在环境的安全策略，只在自己被授权的设备上用
